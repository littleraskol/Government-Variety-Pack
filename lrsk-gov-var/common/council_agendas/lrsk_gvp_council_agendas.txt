# Total Zeal: Age of Wonders (Settled)

agenda_lrsk_gvp_wonder_age = {
	agenda_cost = 7000	# base default
	potential = {
		lrsk_gvp_has_total_zeal = yes
		is_nomadic = no
		NOT = { has_modifier = agenda_lrsk_gvp_wonder_age_finish }
	}
	allow = {
		custom_tooltip = {
			success_text = LRSK_GVP_IS_ABSOLUTE_PASS
			fail_text = LRSK_GVP_IS_ABSOLUTE_FAIL
			is_absolute = yes
		}
		OR = {
			custom_tooltip = {
				success_text = LRSK_GVP_CAN_WONDER_AGE_MEGA
				fail_text = LRSK_GVP_CANT_WONDER_AGE_MEGA
				can_build_megastructures = yes
			}
			custom_tooltip = {
				success_text = LRSK_GVP_CAN_WONDER_AGE_TERRA
				fail_text = LRSK_GVP_CANT_WONDER_AGE_TERRA
				lrsk_gvp_can_terraform = yes
			}
		}
	}
	modifier = {
		pop_amenities_usage_mult = 0.1
		planet_pops_consumer_goods_upkeep_mult = 0.1
		planet_jobs_unity_produces_mult = 0.05
	}
	finish_modifier = agenda_lrsk_gvp_wonder_age_finish
}
